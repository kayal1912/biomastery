:core[Describe sexually transmitted infections:]

- :core[an infection that is transmitted through sexual contact]

:core[Name a pathogen that causes an STI:]

- :core[human immunodeficiency virus (HIV)]

:core[What can HIV do?]

- :core[the infection may lead to AIDS]

:core[Describe the transmission of HIV:]

- :core[blood transfusions ]
- :core[unprotected sex]
- :core[sharing needles ]
- :core[breast feeding ]
- :core[during birth ]

:core[How is the spread of STIs controlled?]

- :core[abstinence ]
- :core[barrier method of contraception ]
- :core[clean/do not share needles ]
- :core[blood screening (with donors)]
- :core[HIV screening (with partners)]
- :core[bottle feeding ]
- :core[sterilise medical equipment ]
